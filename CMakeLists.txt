cmake_minimum_required(VERSION 3.20)
project(hexa_udon_production VERSION 0.1.0 LANGUAGES CXX)
option(HEXA_UDON_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(HEXA_UDON_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "HEXA_UDON_ENABLE_SANITIZERS requires GCC or Clang")
    endif()
endif()
execute_process(
    COMMAND git rev-parse --short=12 HEAD
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE HEXA_UDON_BUILD_FINGERPRINT
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
if(NOT HEXA_UDON_BUILD_FINGERPRINT)
    set(HEXA_UDON_BUILD_FINGERPRINT "unknown")
endif()
find_package(nlohmann_json 3.11.3 CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/CompilerWarnings.cmake)

add_library(hexa_udon_core src/core/map_definition.cpp src/core/model_validation.cpp src/core/types.cpp src/core/problem.cpp)
target_include_directories(hexa_udon_core PUBLIC include)
target_link_libraries(hexa_udon_core PUBLIC nlohmann_json::nlohmann_json)
hexa_udon_enable_warnings(hexa_udon_core)
add_library(hexa_udon_simulator src/simulator/action.cpp src/simulator/simulator.cpp)
target_include_directories(hexa_udon_simulator PUBLIC include)
target_link_libraries(hexa_udon_simulator PUBLIC hexa_udon_core)
hexa_udon_enable_warnings(hexa_udon_simulator)
add_library(hexa_udon_pathfinding src/pathfinding/router.cpp)
target_include_directories(hexa_udon_pathfinding PUBLIC include)
target_link_libraries(hexa_udon_pathfinding PUBLIC hexa_udon_core)
hexa_udon_enable_warnings(hexa_udon_pathfinding)
add_library(hexa_udon_planner src/planner/day_planner.cpp)
target_include_directories(hexa_udon_planner PUBLIC include)
target_link_libraries(hexa_udon_planner PUBLIC hexa_udon_core hexa_udon_pathfinding hexa_udon_simulator)
hexa_udon_enable_warnings(hexa_udon_planner)
add_library(hexa_udon_optimizer src/optimizer/optimizer.cpp)
target_include_directories(hexa_udon_optimizer PUBLIC include)
target_link_libraries(hexa_udon_optimizer PUBLIC hexa_udon_planner Threads::Threads)
hexa_udon_enable_warnings(hexa_udon_optimizer)
add_library(hexa_udon_prematch_selector src/planner/prematch_type_selector.cpp)
target_include_directories(hexa_udon_prematch_selector PUBLIC include)
target_link_libraries(hexa_udon_prematch_selector PUBLIC hexa_udon_optimizer)
hexa_udon_enable_warnings(hexa_udon_prematch_selector)
# procon2026 から移した solver のコードは int と size_t を混ぜて書いているので、型変換と符号と名前の隠れの警告は出さない
set_source_files_properties(src/core/problem.cpp src/simulator/simulator.cpp src/pathfinding/router.cpp
    src/planner/day_planner.cpp src/optimizer/optimizer.cpp src/planner/prematch_type_selector.cpp
    PROPERTIES COMPILE_OPTIONS "-Wno-conversion;-Wno-sign-conversion;-Wno-sign-compare;-Wno-shadow;-Wno-missing-field-initializers")
add_library(hexa_udon_json src/protocol/json_codec.cpp)
target_include_directories(hexa_udon_json PUBLIC include)
target_link_libraries(hexa_udon_json PUBLIC hexa_udon_core hexa_udon_simulator nlohmann_json::nlohmann_json)
hexa_udon_enable_warnings(hexa_udon_json)
add_library(hexa_udon_http src/protocol/http_transport.cpp)
target_include_directories(hexa_udon_http PUBLIC include)
target_link_libraries(hexa_udon_http PUBLIC CURL::libcurl)
hexa_udon_enable_warnings(hexa_udon_http)
add_library(hexa_udon_protocol src/protocol/api_client.cpp src/protocol/request_control.cpp)
target_include_directories(hexa_udon_protocol PUBLIC include)
target_link_libraries(hexa_udon_protocol PUBLIC hexa_udon_json hexa_udon_http nlohmann_json::nlohmann_json)
hexa_udon_enable_warnings(hexa_udon_protocol)
add_library(hexa_udon_session src/session/session.cpp src/session/polling.cpp src/session/persistence.cpp)
target_include_directories(hexa_udon_session PUBLIC include)
target_link_libraries(hexa_udon_session PUBLIC hexa_udon_protocol hexa_udon_simulator nlohmann_json::nlohmann_json)
hexa_udon_enable_warnings(hexa_udon_session)
add_library(hexa_udon_app src/app/auto_client.cpp src/app/session_lock.cpp src/app/lan_worker.cpp)
target_include_directories(hexa_udon_app PUBLIC include)
target_link_libraries(hexa_udon_app PUBLIC hexa_udon_session hexa_udon_prematch_selector)
target_compile_definitions(hexa_udon_app PUBLIC HEXA_UDON_BUILD_FINGERPRINT="${HEXA_UDON_BUILD_FINGERPRINT}")
hexa_udon_enable_warnings(hexa_udon_app)
add_executable(hexa_udon src/app/main.cpp)
target_link_libraries(hexa_udon PRIVATE hexa_udon_app)
hexa_udon_enable_warnings(hexa_udon)

include(CTest)

    add_executable(app_unit_tests tests/unit/app_test.cpp)
    target_link_libraries(app_unit_tests PRIVATE hexa_udon_app)
    hexa_udon_enable_warnings(app_unit_tests)
    add_test(NAME app_unit_tests COMMAND app_unit_tests)

    add_executable(lan_worker_unit_tests tests/unit/lan_worker_test.cpp)
    target_link_libraries(lan_worker_unit_tests PRIVATE hexa_udon_app)

    add_executable(control_tests tests/unit/control_test.cpp)
    target_link_libraries(control_tests PRIVATE hexa_udon_protocol)
    hexa_udon_enable_warnings(control_tests)
    add_test(NAME control_tests COMMAND control_tests)

    add_executable(protocol_tests tests/unit/protocol_test.cpp)
    target_link_libraries(protocol_tests PRIVATE hexa_udon_protocol)
    hexa_udon_enable_warnings(protocol_tests)
    add_test(NAME protocol_tests COMMAND protocol_tests)

    add_executable(session_tests tests/unit/session_test.cpp)
    target_link_libraries(session_tests PRIVATE hexa_udon_session)
    hexa_udon_enable_warnings(session_tests)
    add_test(NAME session_tests COMMAND session_tests)
endif()
